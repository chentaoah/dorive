### 控制器

```java
@Validated
@RestController
@AllArgsConstructor
@RequestMapping("/${module_path}/${entity}")
public class ${Entity}Controller {

    private final ${Entity}Service service;

    /**
     * 新增
     */
    @OrganizationId
    @PostMapping("/add")
    public ResObject<Object> add(@Valid @RequestBody AddDto dto) {
        return service.add(Options.ROOT, dto.toEntity());
    }

    /**
     * 新增（批量）
     */
    @PostMapping("/addBatch")
    public ResObject<Object> addBatch(@Valid @RequestBody List<AddDto> dtoList) {
        return service.addBatch(Options.ROOT, dtoList.stream().map(AddDto::toEntity).collect(Collectors.toList()));
    }

    /**
     * 列表
     */
    @OrganizationId
    @GetMapping("/list")
    public ResObject<List<ListVo>> list(@Valid ListDto dto) {
        List<${Entity}> entities = service.list(Options.ROOT, dto.toQuery());
        List<ListVo> vos = entities.stream().map(ListVo::new).collect(Collectors.toList());
        return ResObject.success(vos);
    }

    /**
     * 列表（分页）
     */
    @OrganizationId
    @GetMapping("/page")
    public ResObject<Page<ListVo>> page(@Valid ListDto dto) {
        Page<${Entity}> page = service.page(Options.ROOT, dto.toQuery());
        List<${Entity}> entities = page.getRecords();
        List<ListVo> vos = entities.stream().map(ListVo::new).collect(Collectors.toList());
        Page<ListVo> newPage = new Page<>();
        newPage.setTotal(page.getTotal());
        newPage.setCurrent(page.getCurrent());
        newPage.setSize(page.getSize());
        newPage.setRecords(vos);
        return ResObject.success(newPage);
    }

    /**
     * 更新
     */
    @OrganizationId
    @PostMapping("/edit")
    public ResObject<Object> edit(@Valid @RequestBody EditDto dto) {
        return service.edit(Options.ROOT, dto.toEntity());
    }

    /**
     * 更新（批量）
     */
    @PostMapping("/editBatch")
    public ResObject<Object> editBatch(@Valid @RequestBody List<EditDto> dtoList) {
        return service.editBatch(Options.ROOT, dtoList.stream().map(EditDto::toEntity).collect(Collectors.toList()));
    }

    /**
     * 删除
     */
    @PostMapping("/delete")
    public ResObject<Object> delete(@NotNull Integer id) {
        return service.delete(Options.ROOT, id);
    }

}
```
